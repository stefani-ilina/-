#include <Servo.h>


const float OFFSET_SENSOR_1 = 5.0;  // Офсет за сензорот на количката
const float OFFSET_SENSOR_2 = 0.0; // Офсет за коцката

#define Umax 40
#define Umin -40
#define Umax_rad 0.698
#define Umin_rad -0.698
#define T 0.09

const int echoPin2 = 4;
const int trigPin2 = 3;
const int echoPin1 = 6;
const int trigPin1 = 7;

Servo servo;

double setpoint, setpoint_prec;
double y, y_prec;
double error;
double P, I, D, U;
double U_prec = 0;
double I_prec = 0;
double D_prec = 0;
boolean Saturation = false;

// PID параметри
double Kp = 6.8; 
double Ki = 1.8;
double Kd = 1.6;

void setup() {
  Serial.begin(9600);

  pinMode(trigPin2, OUTPUT);
  pinMode(echoPin2, INPUT);
  pinMode(trigPin1, OUTPUT);
  pinMode(echoPin1, INPUT);

  servo.attach(9);

  delay(1000);
  move_servo(90); 
  delay(2000);

  // Првични мерења
  setpoint_prec = measure_2();
  y_prec = measure_1();
  delay(500);
}

void loop() {
  double raw_setpoint = measure_2();
  setpoint = 0.53 * raw_setpoint + 0.47 * setpoint_prec;


  double raw_y = measure_1();
  y = 0.53 * raw_y + 0.47 * y_prec;

  error = y - setpoint;

  P = Kp * error;

  if (!Saturation) {
    I = I_prec + T * Ki * error;
  } else {
    I = I_prec; // Anti-windup
  }

  D = (Kd / T) * (y - y_prec);
  D = 0.56 * D + 0.44 * D_prec; 

  U = P + I + D;


  if (U < Umin_rad) {
    U = Umin_rad;
    Saturation = true;
  } else if (U > Umax_rad) {
    U = Umax_rad;
    Saturation = true;
  } else {
    Saturation = false;
  }

  int servo_angle = round(U * 180 / M_PI);
  int final_angle = map(servo_angle, Umin, Umax, 40, 140);

  if (abs(error) > 0.005) { 
    move_servo(final_angle);
  }

  Serial.print("Kocka: "); Serial.print(setpoint * 100); 
  Serial.print("cm | Kolicka: "); Serial.print(y * 100);
  Serial.print("cm | Error: "); Serial.println(error * 100);
  
  I_prec = I;
  y_prec = y;
  D_prec = D;
  setpoint_prec = setpoint;

  delay(20); 
}

float measure_1(void) {
  digitalWrite(trigPin1, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin1, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin1, LOW);

  long duration = pulseIn(echoPin1, HIGH, 25000); 
  float distance = (float)duration / 58.2;

  if (distance > 30) distance = 30;
  if (distance <= 5) distance = 5; 
  
  return 0.01 * (distance - OFFSET_SENSOR_1);
}

float measure_2(void) {
  digitalWrite(trigPin2, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin2, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin2, LOW);

  long duration = pulseIn(echoPin2, HIGH, 25000);
  float distance = (float)duration / 58.2;

  if (distance > 30) distance = 30;
  if (distance <= 2) distance = 2;

  return 0.01 * (distance - OFFSET_SENSOR_2);
}

void move_servo(int u) {
  servo.write(u);
}
